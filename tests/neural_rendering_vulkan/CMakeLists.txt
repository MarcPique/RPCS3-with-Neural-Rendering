cmake_minimum_required(VERSION 3.28)
project(neural_rendering_vulkan_smoke LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(Vulkan REQUIRED)
add_executable(neural_rendering_vulkan_smoke vulkan_smoke.cpp ../../rpcs3/rpcs3qt/neural_rendering_config.cpp)
target_include_directories(neural_rendering_vulkan_smoke PRIVATE ../../rpcs3/rpcs3qt)
target_link_libraries(neural_rendering_vulkan_smoke PRIVATE Qt6::Core Vulkan::Vulkan)
if(MSVC)
  target_compile_options(neural_rendering_vulkan_smoke PRIVATE /utf-8)
endif()
