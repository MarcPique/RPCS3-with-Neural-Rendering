cmake_minimum_required(VERSION 3.28)
project(neural_rendering_config_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Core)
add_executable(neural_rendering_config_tests config_tests.cpp ../../rpcs3/rpcs3qt/neural_rendering_config.cpp)
target_include_directories(neural_rendering_config_tests PRIVATE ../../rpcs3/rpcs3qt)
target_link_libraries(neural_rendering_config_tests PRIVATE Qt6::Core)
if(MSVC)
  target_compile_options(neural_rendering_config_tests PRIVATE /utf-8)
endif()
enable_testing()
add_test(NAME config_round_trip COMMAND neural_rendering_config_tests)
